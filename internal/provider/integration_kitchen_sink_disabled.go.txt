// Hand-written end-to-end integration test that
// exercises the provider against a real device, creating a realistic
// network configuration with proper inter-resource dependencies. Where the
// per-menu acc tests fall short — they're isolated by design — this test
// shows everything works together.
//
// Build tag matches the per-menu tests so `go test -tags acceptance`
// runs both.

//go:build acceptance

package provider

import (
	"fmt"
	"os"
	"testing"

	"github.com/hashicorp/terraform-plugin-testing/helper/resource"
)

// TestAccIntegrationKitchenSink builds a realistic configuration across ~30
// resources with cross-references the dependency-isolated per-menu tests
// can't express:
//
//   bridge -> bridge_port (interface), bridge_vlan
//   ip_pool, ip_address (on bridge), ip_dns_static
//   dhcp_server_network -> dhcp_server (on bridge, using pool), dhcp_server_lease
//   firewall_address_list, firewall_filter (positions), firewall_nat
//   wireguard interface -> wireguard peer (referencing the interface)
//   ip_ipsec_profile -> peer (using profile) -> identity (using peer) -> policy
//   ospf_instance -> ospf_area (using instance)
//   system_script -> system_scheduler (using script)
//   user_group -> user (using group)
//   queue_simple, snmp_community
//   ipv6_pool, ipv6_address, ipv6_firewall_filter (positions)
//
// All in one apply, with terraform managing the dependency graph.
func TestAccIntegrationKitchenSink(t *testing.T) {
	if os.Getenv("TF_ACC") == "" || os.Getenv("ROUTEROS_HOST") == "" {
		t.Skip("TF_ACC and ROUTEROS_HOST required")
	}
	cfg := formatProviderCfg(`
provider "routeros" {
  routers = {
    home = {
      host     = "%s"
      username = "%s"
      password = "%s"
      insecure = true
    }
  }
}

# -------- bridge + child resources --------
resource "routeros_interface_bridge" "br" {
  router = "home"
  name   = "ks-br"
}

resource "routeros_interface_bridge_port" "br_port" {
  router    = "home"
  bridge    = routeros_interface_bridge.br.name
  interface = "ether2"
}

resource "routeros_interface_bridge_vlan" "br_vlan" {
  router   = "home"
  bridge   = routeros_interface_bridge.br.name
  vlan_ids = "100"
  tagged   = routeros_interface_bridge.br.name
}

resource "routeros_interface_vlan" "vlan100" {
  router    = "home"
  name      = "ks-vlan100"
  vlan_id   = 100
  interface = routeros_interface_bridge.br.name
}

# -------- IP layer --------
resource "routeros_ip_pool" "pool" {
  router = "home"
  name   = "ks-pool"
  ranges = "10.70.0.10-10.70.0.250"
}

resource "routeros_ip_address" "lan" {
  router    = "home"
  address   = "10.70.0.1/24"
  interface = routeros_interface_bridge.br.name
}

# -------- DHCP --------
resource "routeros_ip_dhcp_server_network" "net" {
  router  = "home"
  address = "10.70.0.0/24"
  gateway = "10.70.0.1"
}

resource "routeros_ip_dhcp_server" "dhcps" {
  router       = "home"
  name         = "ks-dhcps"
  interface    = routeros_interface_bridge.br.name
  address_pool = routeros_ip_pool.pool.name
  depends_on   = [routeros_ip_dhcp_server_network.net]
}

resource "routeros_ip_dhcp_server_lease" "lease" {
  router      = "home"
  address     = "10.70.0.5"
  mac_address = "02:00:00:00:99:01"
  server      = routeros_ip_dhcp_server.dhcps.name
}

# -------- DNS / lists --------
resource "routeros_ip_dns_static" "rec" {
  router  = "home"
  name    = "ks.router"
  address = "10.70.0.1"
}

resource "routeros_interface_list" "lst" {
  router = "home"
  name   = "KS-LIST"
}

resource "routeros_interface_list_member" "lst_member" {
  router    = "home"
  list      = routeros_interface_list.lst.name
  interface = routeros_interface_bridge.br.name
}

resource "routeros_ip_firewall_address_list" "addrlist" {
  router  = "home"
  list    = "ks-net"
  address = "10.70.0.0/24"
}

# -------- Firewall with positions (deterministic order) --------
resource "routeros_ip_firewall_filter" "fw_allow" {
  router           = "home"
  chain            = "forward"
  action           = "accept"
  comment          = "ks-allow"
  position         = 100
}

resource "routeros_ip_firewall_filter" "fw_drop" {
  router      = "home"
  chain       = "forward"
  action      = "drop"
  comment     = "ks-default-drop"
  position    = 999
  lockout_ack = true
}

resource "routeros_ip_firewall_nat" "nat_masq" {
  router        = "home"
  chain         = "srcnat"
  action        = "masquerade"
  out_interface = "ether1"
  comment       = "ks-masq"
  position      = 100
}

# -------- WireGuard --------
resource "routeros_interface_wireguard" "wg" {
  router      = "home"
  name        = "ks-wg"
  listen_port = 51820
}

resource "routeros_interface_wireguard_peers" "wg_peer" {
  router           = "home"
  interface        = routeros_interface_wireguard.wg.name
  public_key       = "abcdefghijklmnopqrstuvwxyz0123456789ABCDEFG="
  endpoint_address = "203.0.113.1"
  endpoint_port    = 51820
  allowed_address  = "10.99.0.0/24"
}

# -------- OSPF --------
resource "routeros_routing_ospf_instance" "ospf" {
  router    = "home"
  name      = "ks-ospf"
  router_id = "10.70.0.1"
}

resource "routeros_routing_ospf_area" "ospf_area" {
  router   = "home"
  name     = "ks-backbone"
  area_id  = "0.0.0.0"
  instance = routeros_routing_ospf_instance.ospf.name
}

# -------- IPSec full chain --------
resource "routeros_ip_ipsec_profile" "ipp" {
  router         = "home"
  name           = "ks-ipp"
  dh_group       = ["modp2048"]
  enc_algorithm  = ["aes-256"]
  hash_algorithm = "sha256"
}

resource "routeros_ip_ipsec_peer" "ipr" {
  router  = "home"
  name    = "ks-ipr"
  address = "203.0.113.10/32"
  profile = routeros_ip_ipsec_profile.ipp.name
}

resource "routeros_ip_ipsec_proposal" "ipx" {
  router          = "home"
  name            = "ks-ipx"
  auth_algorithms = "sha256"
  enc_algorithms  = ["aes-256-cbc"]
  pfs_group       = "modp2048"
}

resource "routeros_ip_ipsec_identity" "ipid" {
  router = "home"
  peer   = routeros_ip_ipsec_peer.ipr.name
  secret = "ksSharedSecret"
}

resource "routeros_ip_ipsec_policy" "ippol" {
  router      = "home"
  src_address = "10.70.0.0/24"
  dst_address = "10.71.0.0/24"
  peer        = routeros_ip_ipsec_peer.ipr.name
  proposal    = routeros_ip_ipsec_proposal.ipx.name
}

# -------- System script + scheduler --------
resource "routeros_system_script" "scr" {
  router = "home"
  name   = "ks-script"
  source = ":log info ks"
  policy = ["read", "write", "test"]
}

resource "routeros_system_scheduler" "sched" {
  router   = "home"
  name     = "ks-sched"
  on_event = routeros_system_script.scr.name
  interval = "1d"
}

# -------- Queues --------
resource "routeros_queue_simple" "qs" {
  router    = "home"
  name      = "ks-qs"
  target    = "10.70.0.0/24"
  max_limit = "10M/10M"
}

resource "routeros_queue_tree" "qt" {
  router    = "home"
  name      = "ks-qt"
  parent    = "global"
  max_limit = 100000000
}

# -------- Users --------
resource "routeros_user_group" "grp" {
  router = "home"
  name   = "ks-grp"
  policy = ["read", "write"]
}

resource "routeros_user" "usr" {
  router   = "home"
  name     = "ks-user"
  password = "ksUserPw"
  group    = routeros_user_group.grp.name
}

# -------- IPv6 --------
resource "routeros_ipv6_pool" "v6pool" {
  router        = "home"
  name          = "ks-v6pool"
  prefix        = "fd00:db8:7::/56"
  prefix_length = 64
}

resource "routeros_ipv6_address" "v6addr" {
  router    = "home"
  address   = "fd00:db8:7::1/64"
  interface = routeros_interface_bridge.br.name
}

resource "routeros_ipv6_firewall_filter" "v6fw_allow" {
  router           = "home"
  chain            = "forward"
  action           = "accept"
  comment          = "ks-v6-allow"
  position         = 100
}

# -------- SNMP --------
resource "routeros_snmp_community" "snmp" {
  router    = "home"
  name      = "ks-snmp"
  addresses = "10.70.0.0/24"
}
`)
	resource.Test(t, resource.TestCase{
		ProtoV6ProviderFactories: testAccProviderFactories,
		Steps: []resource.TestStep{
			{
				Config: cfg,
				Check: resource.ComposeAggregateTestCheckFunc(
					resource.TestCheckResourceAttrSet("routeros_interface_bridge.br", "id"),
					resource.TestCheckResourceAttrSet("routeros_interface_bridge_port.br_port", "id"),
					resource.TestCheckResourceAttrSet("routeros_ip_address.lan", "id"),
					resource.TestCheckResourceAttrSet("routeros_ip_dhcp_server.dhcps", "id"),
					resource.TestCheckResourceAttrSet("routeros_ip_dhcp_server_lease.lease", "id"),
					resource.TestCheckResourceAttrSet("routeros_ip_firewall_filter.fw_allow", "id"),
					resource.TestCheckResourceAttr("routeros_ip_firewall_filter.fw_allow", "position", "100"),
					resource.TestCheckResourceAttr("routeros_ip_firewall_filter.fw_drop", "position", "999"),
					resource.TestCheckResourceAttrSet("routeros_interface_wireguard.wg", "id"),
					resource.TestCheckResourceAttrSet("routeros_interface_wireguard_peers.wg_peer", "id"),
					resource.TestCheckResourceAttrSet("routeros_routing_ospf_area.ospf_area", "id"),
					resource.TestCheckResourceAttrSet("routeros_ip_ipsec_identity.ipid", "id"),
					resource.TestCheckResourceAttrSet("routeros_ip_ipsec_policy.ippol", "id"),
					resource.TestCheckResourceAttrSet("routeros_user.usr", "id"),
					resource.TestCheckResourceAttrSet("routeros_ipv6_address.v6addr", "id"),
				),
			},
			// Second apply: must be no-op (zero drift).
			{
				Config:   cfg,
				PlanOnly: true,
			},
		},
	})
	fmt.Println("kitchen-sink ok")
}
